export const API = import.meta.env.VITE_API_URL || "";

async function xfetch(url: string, init?: RequestInit) {
  const r = await fetch(url, init);
  const text = await r.text();
  let j: any = {};
  try { j = text ? JSON.parse(text) : {}; } catch { j = { raw: text }; }
  if (!r.ok) throw new Error(j?.error || r.statusText);
  return j;
}

export const getJson = (path: string) => xfetch(`${API}${path}`);
export const postJson = (path: string, body: any) =>
  xfetch(`${API}${path}`, { method: "POST", headers: { "Content-Type":"application/json" }, body: JSON.stringify(body) });
EOF

import React, { useEffect, useMemo, useState } from "react";
import { API, getJson, postJson } from "../lib/api";

type Profile = {
  user_id: string;
  sex?: "male"|"female"|"other";
  age?: number;
  height_cm?: number;
  weight_kg?: number;
  activity_level?: "sedentary"|"light"|"moderate"|"high";
  training_age_months?: number;
};

type Workout = { id:number; owner_id:string; name:string; goal?:string|null; weeks:number; created_at?:string };
type WorkoutItem = { id:number; day_index:number; order_index:number; exercise_id:number; sets:number|null; reps:string|null; rest_sec:number|null; notes:string|null };
type Exercise = { id:number; name:string };

const GOALS = ["hypertrophy","strength","fat_loss","general_fitness"] as const;
const SPLITS = ["auto","full-body","upper/lower","push/pull/legs"] as const;
const presetEquip = (where:"home"|"gym") => where==="home"
  ? ["bodyweight","dumbbell","bands","kettlebell"]
  : ["barbell","dumbbell","machine","cable","kettlebell","bodyweight"];

export default function WorkoutsPage() {
  const [ownerId, setOwnerId] = useState("");
  const [profile, setProfile] = useState<Profile | null>(null);

  const [goal, setGoal] = useState<(typeof GOALS)[number]>("hypertrophy");
  const [split, setSplit] = useState<(typeof SPLITS)[number]>("auto");
  const [daysPerWeek, setDaysPerWeek] = useState(4);
  const [sessionMinutes, setSessionMinutes] = useState(60);
  const [where, setWhere] = useState<"home"|"gym">("gym");
  const [equipment, setEquipment] = useState<string[]>(presetEquip("gym"));
  const [injuries, setInjuries] = useState("");

  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState("");

  const [workouts, setWorkouts] = useState<Workout[]>([]);
  const [expanded, setExpanded] = useState<number|null>(null);
  const [items, setItems] = useState<Record<number, WorkoutItem[]>>({});
  const [exMap, setExMap] = useState<Record<number, Exercise>>({});

  useEffect(() => {
    (async () => {
      const tryEndpoints = ["/auth/me", "/profile/me", "/me"];
      let uid = localStorage.getItem("user_id") || "";
      for (const p of tryEndpoints) {
        try {
          const j = await getJson(p);
          uid = j?.user?.id || j?.data?.user_id || j?.data?.id || uid;
          if (uid) break;
        } catch {}
      }
      if (!uid) uid = "demo_user_1";
      setOwnerId(uid);
      setProfile({ user_id: uid });
    })();
  }, []);

  useEffect(() => { setEquipment(presetEquip(where)); }, [where]);

  useEffect(() => { (async () => {
    if (!ownerId) return;
    try {
      const j = await getJson(`/workouts?ownerId=${encodeURIComponent(ownerId)}`);
      if (j?.success) setWorkouts(j.data || []);
    } catch {}
  })(); }, [ownerId]);

  const canSubmit = useMemo(() => !!ownerId && !busy, [ownerId, busy]);

  async function generatePlan() {
    setBusy(true); setErr("");
    try {
      const body = {
        ownerId,
        profile,
        prefs: {
          goal,
          days_per_week: daysPerWeek,
          session_minutes: sessionMinutes,
          equipment,
          injuries,
          desired_split: split
        },
        weeks: 8
      };
      const res = await postJson("/workouts/ai", body);
      if (!res?.success) throw new Error(res?.error || "failed");
      // refresh list
      const j = await getJson(`/workouts?ownerId=${encodeURIComponent(ownerId)}`);
      if (j?.success) setWorkouts(j.data || []);
    } catch (e:any) {
      setErr(e?.message || "Failed to create plan");
    } finally {
      setBusy(false);
    }
  }

  async function toggleExpand(id:number) {
    if (expanded === id) { setExpanded(null); return; }
    setExpanded(id);
    if (!items[id]) {
      try {
        const j = await getJson(`/workouts/${id}/items`);
        if (j?.success) {
          const list:WorkoutItem[] = j.data || [];
          // optionally fetch names (if you don’t use /full)
          const map = { ...exMap };
          const unique = Array.from(new Set(list.map(x => x.exercise_id)));
          for (const exId of unique) {
            try {
              const ex = await getJson(`/exercises?id=${exId}`);
              if (ex?.data?.length === 1) map[exId] = ex.data[0];
            } catch {}
          }
          setExMap(map);
          setItems({ ...items, [id]: list });
        }
      } catch {}
    }
  }

  function groupByDay(list:WorkoutItem[]) {
    const g:Record<number,WorkoutItem[]> = {};
    for (const it of list) (g[it.day_index] ||= []).push(it);
    for (const k of Object.keys(g)) g[+k].sort((a,b)=>a.order_index-b.order_index);
    return g;
  }

  return (
    <div className="max-w-5xl mx-auto p-4 space-y-6">
      <h1 className="text-2xl font-semibold">Workout Plans</h1>

      <div className="border rounded-xl p-4 space-y-4">
        <h2 className="text-lg font-medium">Create new AI plan</h2>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div>
            <label className="block text-sm mb-1">Goal</label>
            <select className="w-full border rounded px-3 py-2" value={goal} onChange={e=>setGoal(e.target.value as any)}>
              {GOALS.map(g => <option key={g} value={g}>{g}</option>)}
            </select>
          </div>

          <div>
            <label className="block text-sm mb-1">Split</label>
            <select className="w-full border rounded px-3 py-2" value={split} onChange={e=>setSplit(e.target.value as any)}>
              {SPLITS.map(s => <option key={s} value={s}>{s}</option>)}
            </select>
          </div>

          <div>
            <label className="block text-sm mb-1">Days per week</label>
            <input type="number" min={2} max={7} className="w-full border rounded px-3 py-2" value={daysPerWeek} onChange={e=>setDaysPerWeek(+e.target.value)} />
          </div>

          <div>
            <label className="block text-sm mb-1">Session minutes</label>
            <input type="number" min={30} max={120} className="w-full border rounded px-3 py-2" value={sessionMinutes} onChange={e=>setSessionMinutes(+e.target.value)} />
          </div>

          <div>
            <label className="block text-sm mb-1">Where will you train?</label>
            <div className="flex gap-4">
              {(["home","gym"] as const).map(loc => (
                <label key={loc} className="flex items-center gap-2">
                  <input type="radio" name="loc" checked={where===loc} onChange={()=>setWhere(loc)} />
                  <span className="capitalize">{loc}</span>
                </label>
              ))}
            </div>
          </div>

          <div>
            <label className="block text-sm mb-1">Equipment (auto from location; you can edit)</label>
            <div className="flex flex-wrap gap-2">
              {["barbell","dumbbell","machine","cable","kettlebell","bodyweight","bands"].map(e => {
                const on = equipment.includes(e);
                return (
                  <label key={e} className="inline-flex items-center gap-2 border rounded-full px-3 py-1 text-sm">
                    <input type="checkbox" checked={on} onChange={()=>{
                      setEquipment(prev => on ? prev.filter(x=>x!==e) : [...prev, e]);
                    }}/>
                    <span className="capitalize">{e}</span>
                  </label>
                );
              })}
            </div>
          </div>

          <div className="md:col-span-2">
            <label className="block text-sm mb-1">Injuries / constraints (optional)</label>
            <textarea className="w-full border rounded px-3 py-2" rows={2} value={injuries} onChange={e=>setInjuries(e.target.value)} />
          </div>
        </div>

        {err && <div className="text-red-600 text-sm">{err}</div>}

        <button onClick={generatePlan} disabled={!canSubmit} className="px-4 py-2 rounded bg-black text-white disabled:opacity-50">
          {busy ? "Generating…" : "Generate & Save Plan"}
        </button>
      </div>

      <div className="border rounded-xl p-4">
        <h2 className="text-lg font-medium mb-3">My Plans</h2>
        {workouts.length === 0 ? <div className="text-sm text-gray-500">No plans yet.</div> : (
          <div className="space-y-3">
            {workouts.map(w => (
              <div key={w.id} className="border rounded-lg p-3">
                <div className="flex items-center justify-between">
                  <div>
                    <div className="font-medium">{w.name}</div>
                    <div className="text-xs text-gray-500">{w.goal || "—"} • {w.weeks} weeks</div>
                  </div>
                  <button className="px-3 py-1.5 rounded bg-gray-900 text-white text-sm" onClick={()=>toggleExpand(w.id)}>
                    {expanded===w.id ? "Hide" : "View items"}
                  </button>
                </div>

                {expanded===w.id && items[w.id] && (
                  <div className="mt-3">
                    {Object.entries(groupByDay(items[w.id]!)).map(([day, rows]) => (
                      <div key={day} className="mb-2">
                        <div className="font-medium text-sm mb-1">Day {Number(day)+1}</div>
                        <ol className="pl-4 list-decimal space-y-1">
                          {rows.map(it => (
                            <li key={it.id} className="text-sm">
                              <span className="font-medium">
                                {exMap[it.exercise_id]?.name || `#${it.exercise_id}`}
                              </span>
                              {it.sets ? ` — ${it.sets} x ${it.reps || "?"}` : ""}
                              {it.rest_sec ? ` • rest ${it.rest_sec}s` : ""}
                              {it.notes ? ` • ${it.notes}` : ""}
                            </li>
                          ))}
                        </ol>
                      </div>
                    ))}
                  </div>
                )}
              </div>
            ))}
          </div>
        )}
      </div>

      <div className="text-xs text-gray-500">API base: {API || "(not set)"} • owner: {ownerId || "?"}</div>
    </div>
  );
}
