3) Reusable API client

Create src/lib/api.ts:

// src/lib/api.ts
import { API_BASE_URL } from "./config";
import type {
  SupplementRequest, SupplementResponse,
  FitnessChatRequest, FitnessChatResponse
} from "../types/ai";

async function safeFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${API_BASE_URL}${path}`, {
    method: "POST",
    headers: { "Content-Type": "application/json", ...(init?.headers || {}) },
    ...init,
  });

  if (!res.ok) {
    // Try to read FastAPI error detail
    let detail = "";
    try {
      const body = await res.json();
      detail = body?.detail || JSON.stringify(body);
    } catch {
      detail = await res.text();
    }
    throw new Error(`API ${path} ${res.status}: ${detail || "Unknown error"}`);
  }

  return res.json() as Promise<T>;
}

// ---- endpoints ----

export function suggestSupplements(payload: SupplementRequest) {
  return safeFetch<SupplementResponse>("/ai/supplements/suggest", {
    body: JSON.stringify(payload),
  });
}

export function fitnessChat(payload: FitnessChatRequest) {
  return safeFetch<FitnessChatResponse>("/ai/fitness/chat", {
    body: JSON.stringify(payload),
  });
}

import React, { useState } from "react";
import { suggestSupplements } from "../lib/api";
import type { SupplementRequest } from "../types/ai";

const defaultCtx = {
  goal: "",
  age: undefined as number | undefined,
  sex: "" as string | undefined,
  dietary_prefs: [] as string[],
  region: "IN",
  budget_tier: "mid",
};

export default function SupplementsSuggestForm() {
  const [question, setQuestion] = useState("What could help general recovery and joint health?");
  const [ctx, setCtx] = useState(defaultCtx);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [result, setResult] = useState<null | {
    plan_summary: string;
    what_to_avoid: string[];
    interactions_notes: string;
    disclaimer: string;
  }>(null);

  const onSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);
    setResult(null);
    setLoading(true);
    try {
      const payload: SupplementRequest = {
        question,
        context: {
          goal: ctx.goal || "general wellness",
          age: ctx.age ? Number(ctx.age) : undefined,
          sex: ctx.sex || undefined,
          dietary_prefs: ctx.dietary_prefs?.filter(Boolean),
          region: ctx.region || undefined,
          budget_tier: ctx.budget_tier || undefined,
        },
      };
      const res = await suggestSupplements(payload);
      setResult({
        plan_summary: res.plan_summary,
        what_to_avoid: res.what_to_avoid,
        interactions_notes: res.interactions_notes,
        disclaimer: res.disclaimer,
      });
    } catch (err: any) {
      setError(err.message || "Request failed");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-3xl mx-auto p-4 space-y-4">
      <h2 className="text-2xl font-semibold">Supplements — Suggestions (MVP)</h2>

      <form onSubmit={onSubmit} className="space-y-3 bg-white/70 p-4 rounded-2xl shadow">
        <label className="block">
          <span className="text-sm font-medium">Question</span>
          <textarea
            className="mt-1 w-full rounded-xl border p-2"
            value={question}
            onChange={(e) => setQuestion(e.target.value)}
            rows={3}
          />
        </label>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
          <label className="block">
            <span className="text-sm font-medium">Goal</span>
            <input
              className="mt-1 w-full rounded-xl border p-2"
              placeholder="muscle gain / recovery / sleep"
              value={ctx.goal}
              onChange={(e) => setCtx((s) => ({ ...s, goal: e.target.value }))}
            />
          </label>
          <label className="block">
            <span className="text-sm font-medium">Age</span>
            <input
              type="number"
              className="mt-1 w-full rounded-xl border p-2"
              value={ctx.age ?? ""}
              onChange={(e) => setCtx((s) => ({ ...s, age: e.target.value ? Number(e.target.value) : undefined }))}
            />
          </label>
          <label className="block">
            <span className="text-sm font-medium">Sex</span>
            <input
              className="mt-1 w-full rounded-xl border p-2"
              placeholder="male / female / other"
              value={ctx.sex || ""}
              onChange={(e) => setCtx((s) => ({ ...s, sex: e.target.value }))}
            />
          </label>
          <label className="block">
            <span className="text-sm font-medium">Dietary Prefs</span>
            <input
              className="mt-1 w-full rounded-xl border p-2"
              placeholder="comma separated, e.g. vegetarian, dairy-free"
              onChange={(e) =>
                setCtx((s) => ({ ...s, dietary_prefs: e.target.value.split(",").map((x) => x.trim()).filter(Boolean) }))
              }
            />
          </label>
          <label className="block">
            <span className="text-sm font-medium">Region</span>
            <input
              className="mt-1 w-full rounded-xl border p-2"
              value={ctx.region}
              onChange={(e) => setCtx((s) => ({ ...s, region: e.target.value }))}
            />
          </label>
          <label className="block">
            <span className="text-sm font-medium">Budget Tier</span>
            <input
              className="mt-1 w-full rounded-xl border p-2"
              placeholder="value / mid / premium"
              value={ctx.budget_tier}
              onChange={(e) => setCtx((s) => ({ ...s, budget_tier: e.target.value }))}
            />
          </label>
        </div>

        <button
          type="submit"
          className="rounded-xl px-4 py-2 bg-black text-white disabled:opacity-50"
          disabled={loading}
        >
          {loading ? "Thinking..." : "Get Suggestions"}
        </button>
      </form>

      {error && <div className="text-red-600 text-sm">{error}</div>}

      {result && (
        <div className="space-y-3 p-4 rounded-2xl bg-gray-50 border">
          <div className="whitespace-pre-wrap text-sm leading-6">{result.plan_summary}</div>
          <div>
            <h3 className="font-medium">What to avoid</h3>
            <ul className="list-disc pl-6 text-sm">
              {result.what_to_avoid.map((w, i) => <li key={i}>{w}</li>)}
            </ul>
          </div>
          <div className="text-xs text-gray-500">{result.interactions_notes}</div>
          <div className="text-xs text-amber-700">{result.disclaimer}</div>
        </div>
      )}
    </div>
  );

5) UI: Fitness Chat Widget

Create src/components/FitnessChatWidget.tsx:

import React, { useState } from "react";
import { fitnessChat } from "../lib/api";
import type { FitnessChatRequest } from "../types/ai";

export default function FitnessChatWidget() {
  const [input, setInput] = useState("");
  const [messages, setMessages] = useState<{ role: "user" | "assistant"; content: string }[]>([]);
  const [loading, setLoading] = useState(false);
  const [lastDisclaimer, setLastDisclaimer] = useState<string | null>(null);

  const send = async () => {
    if (!input.trim()) return;
    const userMsg = input.trim();
    setMessages((m) => [...m, { role: "user", content: userMsg }]);
    setInput("");
    setLoading(true);

    try {
      const payload: FitnessChatRequest = {
        user_message: userMsg,
        user_profile: undefined,   // optionally pass profile object
        attachments: undefined     // or [{kind:'text', content:'...'}]
      };
      const res = await fitnessChat(payload);
      setMessages((m) => [...m, { role: "assistant", content: res.reply }]);
      setLastDisclaimer(res.disclaimer || null);
    } catch (err: any) {
      setMessages((m) => [...m, { role: "assistant", content: `Error: ${err.message || "failed"}` }]);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-2xl mx-auto p-4 space-y-3">
      <h2 className="text-2xl font-semibold">Fitness AI Chat</h2>

      <div className="h-64 overflow-y-auto border rounded-2xl p-3 bg-white">
        {messages.length === 0 && <div className="text-sm text-gray-500">Ask anything about training, recovery, sleep, nutrition fundamentals.</div>}
        {messages.map((m, i) => (
          <div key={i} className={`my-2 ${m.role === "user" ? "text-right" : "text-left"}`}>
            <div className={`inline-block px-3 py-2 rounded-2xl text-sm ${
              m.role === "user" ? "bg-black text-white" : "bg-gray-100"
            }`}>
              {m.content}
            </div>
          </div>
        ))}
      </div>

      <div className="flex gap-2">
        <input
          className="flex-1 rounded-xl border p-2"
          placeholder="How can I improve recovery after leg day?"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" ? send() : undefined}
        />
        <button
          onClick={send}
          disabled={loading}
          className="rounded-xl px-4 py-2 bg-black text-white disabled:opacity-50"
        >
          {loading ? "..." : "Send"}
        </button>
      </div>

      {lastDisclaimer && <div className="text-xs text-amber-700">{lastDisclaimer}</div>}
    </div>
  );
}

Replit tips

Put VITE_API_BASE_URL in Secrets; Replit will expose it at runtime.

If your FastAPI is running on a different origin and you see CORS errors:

In FastAPI, set CORS_ALLOW_ORIGINS to your Replit URL (comma-separated) and restart.

For local dev: run FastAPI on http://localhost:8000
