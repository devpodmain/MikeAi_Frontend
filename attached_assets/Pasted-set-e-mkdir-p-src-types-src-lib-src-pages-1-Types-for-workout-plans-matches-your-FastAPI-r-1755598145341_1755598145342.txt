import type { WorkoutPlan } from "@/types/workoutPlan";
import type { Profile as MealProfile } from "@/types/mealPlan";

const API_BASE = import.meta.env.VITE_API_BASE_URL ?? "http://localhost:8000";

export class WorkoutPlanApiError extends Error {
  constructor(message: string, public status?: number) {
    super(message);
    this.name = "WorkoutPlanApiError";
  }
}

async function fetchWithErrorHandling(url: string, options: RequestInit = {}) {
  const response = await fetch(url, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(options.headers || {})
    }
  });

  if (!response.ok) {
    let msg = `HTTP ${response.status}: ${response.statusText}`;
    try {
      const err = await response.json();
      msg = err.detail || err.message || msg;
    } catch {}
    throw new WorkoutPlanApiError(msg, response.status);
  }
  return response.json();
}

/** Map DB profile -> workout profile expected by FastAPI */
export function mapDbProfileToWorkout(profile: MealProfile) {
  // DB fields you listed:
  // user_id, gender -> sex, age, height -> height_cm, weight -> weight_kg,
  // activity_level, training_age_months (optional default=6)
  const height_cm =
    (profile as any).height_cm ??
    (profile as any).height ??
    undefined;
  const weight_kg =
    (profile as any).weight_kg ??
    (profile as any).weight ??
    undefined;

  return {
    user_id: profile.user_id,
    sex: (profile as any).sex ?? (profile as any).gender ?? undefined,
    age: (profile as any).age ?? undefined,
    height_cm: height_cm !== undefined ? Number(height_cm) : undefined,
    weight_kg: weight_kg !== undefined ? Number(weight_kg) : undefined,
    activity_level: (profile as any).activity_level ?? undefined,
    training_age_months:
      (profile as any).training_age_months !== undefined
        ? Number((profile as any).training_age_months)
        : 6
  };
}

export type WorkoutPrefs = {
  goal: "hypertrophy" | "strength" | "fat_loss" | "general_fitness";
  days_per_week: number;          // required
  session_minutes?: number;       // optional per your requirement
  equipment?: string[];           // optional per your requirement
  injuries?: string;
  desired_split?: "auto" | "full-body" | "upper/lower" | "push/pull/legs";
};

/** Generate a workout plan via your local FastAPI */
export async function generateWorkoutPlan(
  dbProfile: MealProfile,
  prefs: WorkoutPrefs,
  weeks: number = 8
): Promise<WorkoutPlan> {
  // Build prefs payload with only provided optionals
  const prefsPayload: any = {
    goal: prefs.goal,
    days_per_week: prefs.days_per_week,
  };
  if (prefs.session_minutes !== undefined) prefsPayload.session_minutes = prefs.session_minutes;
  if (prefs.equipment !== undefined) prefsPayload.equipment = prefs.equipment;
  if (prefs.injuries) prefsPayload.injuries = prefs.injuries;
  if (prefs.desired_split) prefsPayload.desired_split = prefs.desired_split;

  const body = {
    profile: mapDbProfileToWorkout(dbProfile),
    prefs: prefsPayload,
    weeks
  };

  try {
    const data = await fetchWithErrorHandling(`${API_BASE}/ai/workouts/generate`, {
      method: "POST",
      body: JSON.stringify(body)
    });
    if (!data?.success || !data?.data) {
      throw new WorkoutPlanApiError("Failed to generate workout plan");
    }
    return data.data as WorkoutPlan;
  } catch (e) {
    if (e instanceof WorkoutPlanApiError) throw e;
    throw new WorkoutPlanApiError("Network error while generating workout plan");
  }
}

/** Local fallback persistence like meals */
export async function persistWorkoutPlan(userId: string, plan: WorkoutPlan): Promise<void> {
  try {
    localStorage.setItem(`workout-plan-${userId}`, JSON.stringify(plan));
  } catch {
    throw new WorkoutPlanApiError("Failed to persist workout plan");
  }
}

export async function loadExistingWorkoutPlan(userId: string): Promise<WorkoutPlan | null> {
  try {
    const raw = localStorage.getItem(`workout-plan-${userId}`);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
}

import React, { useEffect, useMemo, useState } from "react";
import { type WorkoutPlan } from "@/types/workoutPlan";
import { generateWorkoutPlan, persistWorkoutPlan, loadExistingWorkoutPlan, type WorkoutPrefs } from "@/lib/workoutPlanApi";
import { getUserProfile } from "@/lib/mealPlanApi"; // already points to /api/db/get-profile
import type { Profile } from "@/types/mealPlan";

const EQUIP_HOME = ["bodyweight","dumbbell","bands","kettlebell"];
const EQUIP_GYM  = ["barbell","dumbbell","machine","cable","kettlebell","bodyweight"];

export default function WorkoutsPage() {
  const [userId, setUserId] = useState<string>("demo_user_1");
  const [profile, setProfile] = useState<Profile | null>(null);

  // Form state
  const [goal, setGoal] = useState<WorkoutPrefs["goal"]>("hypertrophy");
  const [split, setSplit] = useState<NonNullable<WorkoutPrefs["desired_split"]>>("auto");
  const [daysPerWeek, setDaysPerWeek] = useState<number>(4);

  // Optional fields per your spec
  const [sessionMinutes, setSessionMinutes] = useState<number | undefined>(undefined);
  const [where, setWhere] = useState<"home"|"gym">("gym");
  const [equipment, setEquipment] = useState<string[] | undefined>(EQUIP_GYM);
  const [injuries, setInjuries] = useState<string>("");

  const [plan, setPlan] = useState<WorkoutPlan | null>(null);
  const [loading, setLoading] = useState(false);
  const [err, setErr] = useState("");

  useEffect(() => {
    (async () => {
      try {
        // If your app knows the user id elsewhere, set it here.
        const uid = localStorage.getItem("user_id") || "demo_user_1";
        setUserId(uid);
        const dbProf = await getUserProfile(uid); // hits /api/db/get-profile
        setProfile(dbProf as Profile);

        const stored = await loadExistingWorkoutPlan(uid);
        if (stored) setPlan(stored);
      } catch {
        setProfile({ user_id: "demo_user_1" } as Profile);
      }
    })();
  }, []);

  // Auto equipment by location; user can clear it to make it "optional"
  useEffect(() => {
    setEquipment(where === "home" ? EQUIP_HOME : EQUIP_GYM);
  }, [where]);

  const canSubmit = useMemo(() => !!profile && !loading && !!daysPerWeek && !!goal, [profile, loading, daysPerWeek, goal]);

  async function onGenerate() {
    if (!profile) return;
    setLoading(true); setErr("");

    // Build prefs with optionals only if present
    const prefs: WorkoutPrefs = {
      goal,
      days_per_week: daysPerWeek,
      ...(sessionMinutes !== undefined ? { session_minutes: sessionMinutes } : {}),
      ...(equipment !== undefined ? { equipment } : {}),
      ...(injuries ? { injuries } : {}),
      ...(split ? { desired_split: split } : {})
    };

    try {
      const newPlan = await generateWorkoutPlan(profile, prefs, 8);
      setPlan(newPlan);
      await persistWorkoutPlan(userId, newPlan); // local fallback
    } catch (e: any) {
      setErr(e?.message || "Failed to generate workout plan");
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="max-w-5xl mx-auto p-4 space-y-6">
      <h1 className="text-2xl font-semibold">Workout Plans (AI)</h1>

      {/* Create form */}
      <div className="border rounded-xl p-4 space-y-4">
        <h2 className="text-lg font-medium">Create a plan</h2>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div>
            <label className="block text-sm mb-1">Goal</label>
            <select className="w-full border rounded px-3 py-2" value={goal} onChange={e=>setGoal(e.target.value as any)}>
              {["hypertrophy","strength","fat_loss","general_fitness"].map(g => <option key={g} value={g}>{g}</option>)}
            </select>
          </div>

          <div>
            <label className="block text-sm mb-1">Preferred split</label>
            <select className="w-full border rounded px-3 py-2" value={split} onChange={e=>setSplit(e.target.value as any)}>
              {["auto","full-body","upper/lower","push/pull/legs"].map(s => <option key={s} value={s}>{s}</option>)}
            </select>
          </div>

          <div>
            <label className="block text-sm mb-1">Days per week</label>
            <input type="number" min={2} max={7} className="w-full border rounded px-3 py-2" value={daysPerWeek} onChange={e=>setDaysPerWeek(+e.target.value)} />
          </div>

          <div>
            <label className="block text-sm mb-1">Session minutes (optional)</label>
            <input
              type="number" min={30} max={120} placeholder="Leave empty"
              className="w-full border rounded px-3 py-2"
              value={sessionMinutes ?? ""}
              onChange={e=>{
                const v = e.target.value;
                setSessionMinutes(v === "" ? undefined : +v);
              }}
            />
          </div>

          <div>
            <label className="block text-sm mb-1">Where do you train?</label>
            <div className="flex gap-4">
              {(["home","gym"] as const).map(loc => (
                <label key={loc} className="flex items-center gap-2">
                  <input type="radio" name="loc" checked={where===loc} onChange={()=>setWhere(loc)} />
                  <span className="capitalize">{loc}</span>
                </label>
              ))}
            </div>
            <div className="text-xs text-gray-500 mt-1">You can clear equipment below to keep it optional.</div>
          </div>

          <div className="md:col-span-2">
            <label className="block text-sm mb-1">Equipment (optional)</label>
            <div className="flex flex-wrap gap-2 mb-2">
              {["barbell","dumbbell","machine","cable","kettlebell","bodyweight","bands"].map(e => {
                const on = (equipment || []).includes(e);
                return (
                  <label key={e} className="inline-flex items-center gap-2 border rounded-full px-3 py-1 text-sm">
                    <input
                      type="checkbox"
                      checked={on}
                      onChange={()=>{
                        setEquipment(prev => {
                          const cur = prev ? [...prev] : [];
                          const i = cur.indexOf(e);
                          if (i >= 0) cur.splice(i,1); else cur.push(e);
                          return cur;
                        });
                      }}
                    />
                    <span className="capitalize">{e}</span>
                  </label>
                );
              })}
            </div>
            <button
              type="button"
              className="text-xs underline"
              onClick={()=>setEquipment(undefined)}
              title="Remove equipment from request to keep it optional"
            >
              Clear equipment (make optional)
            </button>
          </div>

          <div className="md:col-span-2">
            <label className="block text-sm mb-1">Injuries / constraints (optional)</label>
            <textarea className="w-full border rounded px-3 py-2" rows={2} value={injuries} onChange={e=>setInjuries(e.target.value)} />
          </div>
        </div>

        {err && <div className="text-red-600 text-sm">{err}</div>}

        <button onClick={onGenerate} disabled={!canSubmit} className="px-4 py-2 rounded bg-black text-white disabled:opacity-50">
          {loading ? "Generating…" : "Generate Plan"}
        </button>
      </div>

      {/* Render plan from backend (no static plans) */}
      {plan && (
        <div className="border rounded-xl p-4">
          <h2 className="text-lg font-medium mb-3">{plan.name} • {plan.goal} • {plan.weeks} weeks</h2>
          {plan.days.sort((a,b)=>a.dayIndex-b.dayIndex).map(d => (
            <div key={d.dayIndex} className="mb-3">
              <div className="font-medium text-sm mb-1">Day {d.dayIndex+1} — {d.name}</div>
              <ol className="pl-4 list-decimal space-y-1">
                {d.items.map((it, idx) => (
                  <li key={idx} className="text-sm">
                    <span className="font-medium">{it.exercise}</span>
                    {` — ${it.sets} x ${it.reps}`}
                    {it.restSec ? ` • rest ${it.restSec}s` : ""}
                    {it.notes ? ` • ${it.notes}` : ""}
                  </li>
                ))}
              </ol>
            </div>
          ))}
          {plan.progression_notes && <div className="text-xs text-gray-600 mt-2">Progression: {plan.progression_notes}</div>}
          {plan.warmup_notes && <div className="text-xs text-gray-600">Warm-up: {plan.warmup_notes}</div>}
          {plan.deload && <div className="text-xs text-gray-600">Deload: {plan.deload}</div>}
        </div>
      )}
    </div>
  );
}
